# Standalone reproducibility script for SpatChat LoCoH outputs
# Install requirements first:
#   pip install pandas numpy shapely scipy pyproj
# Rename this file to .py if you want to run it directly.
#
# USER ARGUMENTS:
# - INPUT_CSV: by default uses bundled input_data.csv in this folder.
# - OUTPUT_DIR: where reproduced outputs will be written.
# - PARAMS: LoCoH settings. By default these come from repro_config.json.

import json
import os
import zipfile
from dataclasses import dataclass
from typing import Optional

import numpy as np
import pandas as pd
from pyproj import Transformer
from scipy.spatial import cKDTree
from shapely.geometry import MultiPoint, Polygon, mapping
from shapely.ops import transform as shp_transform

SCRIPT_DIR = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()
CONFIG_PATH = os.path.join(SCRIPT_DIR, "repro_config.json")
INPUT_CSV = os.path.join(SCRIPT_DIR, "input_data.csv")
OUTPUT_DIR = os.path.join(SCRIPT_DIR, "reproduced_outputs")

CONFIG = {}
if os.path.exists(CONFIG_PATH):
    with open(CONFIG_PATH, "r", encoding="utf-8") as f:
        CONFIG = json.load(f)


@dataclass
class LoCoHParams:
    method: str = "k"
    k: int = 10
    a: Optional[float] = None
    r: Optional[float] = None
    isopleths: tuple[int, ...] = (50, 95)


PARAMS_RAW = CONFIG.get("locoh", {}).get("params", {}) or {}
PARAMS = LoCoHParams(
    method=str(PARAMS_RAW.get("method", "k")),
    k=int(PARAMS_RAW.get("k", 10)),
    a=float(PARAMS_RAW["a"]) if PARAMS_RAW.get("a") is not None else None,
    r=float(PARAMS_RAW["r"]) if PARAMS_RAW.get("r") is not None else None,
    isopleths=tuple(int(x) for x in PARAMS_RAW.get("isopleths", [50, 95])),
)

if not os.path.exists(INPUT_CSV):
    raise FileNotFoundError(f"Bundled input data not found: {INPUT_CSV}")

os.makedirs(OUTPUT_DIR, exist_ok=True)

df = pd.read_csv(INPUT_CSV)
required = {"animal_id", "longitude", "latitude"}
missing = sorted(required - set(df.columns))
if missing:
    raise ValueError(f"Bundled input_data.csv is missing required columns: {missing}")


def compute_locoh(dataframe, params=PARAMS):
    method = params.method.lower()
    if method not in {"k", "a", "r"}:
        raise ValueError("LoCoH method must be one of {'k','a','r'}")
    if method == "a" and (params.a is None or float(params.a) <= 0):
        raise ValueError("a-LoCoH requires a positive 'a' value in meters.")
    if method == "r" and (params.r is None or float(params.r) <= 0):
        raise ValueError("r-LoCoH requires a positive 'r' value in meters.")

    gdf = dataframe[["animal_id", "longitude", "latitude"]].dropna().copy()
    gdf.columns = ["animal_id", "lon", "lat"]

    to_eq = Transformer.from_crs(4326, 6933, always_xy=True)
    to_wgs = Transformer.from_crs(6933, 4326, always_xy=True)

    def reproj_geom(geom):
        return shp_transform(lambda x, y, z=None: to_wgs.transform(x, y), geom)

    xs, ys = to_eq.transform(gdf["lon"].values, gdf["lat"].values)
    gdf["x_m"] = xs
    gdf["y_m"] = ys

    out = {"method": method, "isopleths": list(params.isopleths), "animals": {}}

    for animal, sub in gdf.groupby("animal_id"):
        sub = sub.reset_index(drop=True)
        if len(sub) < 3:
            out["animals"][str(animal)] = {"n_points": int(len(sub)), "isopleths": [], "facets": []}
            continue

        coords_m = np.column_stack([sub["x_m"].values, sub["y_m"].values])
        tree = cKDTree(coords_m)
        hulls = []
        n = len(coords_m)
        k_cap = max(3, min(params.k if method == "k" else 10, n, 50))

        for i, point in enumerate(coords_m):
            if method == "k":
                k = max(3, min(params.k, n, 50))
                _, idxs = tree.query(point, k)
                idxs = np.atleast_1d(idxs)
                pts = coords_m[idxs]
            elif method == "r":
                idxs = tree.query_ball_point(point, float(params.r))
                if len(idxs) < 3:
                    _, idxs = tree.query(point, 3)
                    idxs = np.atleast_1d(idxs)
                pts = coords_m[idxs]
            else:
                dists, idxs = tree.query(point, k=min(max(10, min(2 * k_cap, n)), n))
                dists = np.atleast_1d(dists)
                idxs = np.atleast_1d(idxs)
                order = np.argsort(dists)
                cumdist = 0.0
                kept = []
                for j in order:
                    cumdist += float(dists[j])
                    kept.append(int(idxs[j]))
                    if cumdist >= float(params.a):
                        break
                if len(kept) < 3:
                    _, extra = tree.query(point, 3)
                    kept = list(np.unique(np.atleast_1d(extra)))
                pts = coords_m[kept]

            if len(pts) >= 3:
                hull = MultiPoint(pts).convex_hull
                if isinstance(hull, Polygon):
                    hulls.append(hull)

        if not hulls:
            out["animals"][str(animal)] = {"n_points": int(len(sub)), "isopleths": [], "facets": []}
            continue

        hulls.sort(key=lambda h: h.area)
        total_area = sum(h.area for h in hulls) or 1.0
        facets = []
        cum = 0.0
        for hull in hulls:
            cum += hull.area
            frac = min(1.0, cum / total_area)
            facets.append({
                "cum_frac": float(frac),
                "cum_percent": int(round(frac * 100)),
                "area_sq_km": float(hull.area / 1e6),
                "geometry": mapping(reproj_geom(hull)),
            })

        targets = sorted({int(t) for t in params.isopleths if 1 <= int(t) <= 100})
        polys_by_iso = {}
        cum_union = None
        cum_area = 0.0
        t_idx = 0
        target_fracs = [t / 100.0 for t in targets]

        for hull in hulls:
            cum_area += hull.area
            cum_union = hull if cum_union is None else cum_union.union(hull)
            while t_idx < len(target_fracs) and (cum_area / total_area) >= target_fracs[t_idx]:
                polys_by_iso[targets[t_idx]] = cum_union
                t_idx += 1
            if t_idx >= len(target_fracs):
                break

        if targets and targets[-1] not in polys_by_iso and cum_union is not None:
            polys_by_iso[targets[-1]] = cum_union

        envelopes = []
        for iso, geom_m in polys_by_iso.items():
            envelopes.append({
                "isopleth": int(iso),
                "area_sq_km": float(geom_m.area / 1e6),
                "geometry": mapping(reproj_geom(geom_m)),
            })

        out["animals"][str(animal)] = {
            "n_points": int(len(sub)),
            "isopleths": envelopes,
            "facets": facets,
        }

    return out


result = compute_locoh(df, params=PARAMS)
with open(os.path.join(OUTPUT_DIR, "locoh_results.json"), "w", encoding="utf-8") as f:
    json.dump(result, f)

rows = []
envelope_features = []
facets_features = []

for animal_id, data in (result.get("animals") or {}).items():
    for item in data.get("isopleths", []):
        iso = int(item.get("isopleth"))
        area = float(item.get("area_sq_km", 0.0))
        rows.append((animal_id, f"LoCoH-{iso}", area))
        if item.get("geometry"):
            envelope_features.append({
                "type": "Feature",
                "properties": {"animal_id": str(animal_id), "isopleth": iso, "area_km2": area},
                "geometry": item["geometry"],
            })

    for facet in data.get("facets", []) or []:
        facets_features.append({
            "type": "Feature",
            "properties": {
                "animal_id": str(animal_id),
                "cum_percent": int(facet.get("cum_percent", 0)),
                "area_km2": float(facet.get("area_sq_km", 0.0)),
            },
            "geometry": facet.get("geometry"),
        })

if rows:
    areas = pd.DataFrame(rows, columns=["animal_id", "type", "area_km2"]).sort_values(["animal_id", "type"])
    areas.to_csv(os.path.join(OUTPUT_DIR, "home_range_areas.csv"), index=False)

if envelope_features:
    with open(os.path.join(OUTPUT_DIR, "locoh_envelopes_all.geojson"), "w", encoding="utf-8") as f:
        json.dump({"type": "FeatureCollection", "features": envelope_features}, f)

if facets_features:
    with open(os.path.join(OUTPUT_DIR, "locoh_facets_all.geojson"), "w", encoding="utf-8") as f:
        json.dump({"type": "FeatureCollection", "features": facets_features}, f)

archive_path = os.path.join(OUTPUT_DIR, "spatchat_results.zip")
if os.path.exists(archive_path):
    os.remove(archive_path)
with zipfile.ZipFile(archive_path, "w", zipfile.ZIP_DEFLATED) as zipf:
    for root, _, files in os.walk(OUTPUT_DIR):
        for file in files:
            if file.endswith(".zip"):
                continue
            full_path = os.path.join(root, file)
            rel_path = os.path.relpath(full_path, OUTPUT_DIR)
            zipf.write(full_path, arcname=rel_path)

print(f"LoCoH outputs written to: {OUTPUT_DIR}")
print(f"Zip archive: {archive_path}")
